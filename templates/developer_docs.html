Drop-in component for v0.dev, Bolt.new, Lovable, Next.js (App / Pages router), and Vite React:

'use client';

import React, { useEffect } from 'react';

declare global {
  interface Window {
    RVSSAHWidget: any;
  }
}

interface RvsSahButtonProps {
  apiKey: string;
  onSuccess: (data: { count: number; headers: string[]; students: string[][] }) => void;
  className?: string;
  buttonText?: string;
}

export function RvsSahImportButton({
  apiKey,
  onSuccess,
  className = "px-4 py-2 bg-blue-600 hover:bg-blue-700 text-white font-medium rounded-lg shadow-sm transition",
  buttonText = "Import Students with RVS SAH"
}: RvsSahButtonProps) {
  useEffect(() => {
    const scriptId = 'rvs-sah-sdk';
    if (!document.getElementById(scriptId)) {
      const script = document.createElement('script');
      script.id = scriptId;
      script.src = 'http://localhost:8085/static/widget/widget.js';
      script.async = true;
      script.onload = () => bindWidget();
      document.body.appendChild(script);
    } else {
      bindWidget();
    }

    function bindWidget() {
      if (window.RVSSAHWidget) {
        window.RVSSAHWidget.init({
          apiKey: apiKey,
          triggerElement: '#rvs-sah-import-trigger',
          mode: 'modal',
          onSuccess: (res: any) => onSuccess(res),
          onError: (err: any) => console.error('RVS SAH Error:', err)
        });
      }
    }
  }, [apiKey, onSuccess]);

  return (
    <button id="rvs-sah-import-trigger" type="button" className={className}>
      {buttonText}
    </button>
  );
}

Direct headless Node.js backend automation using axios:

const axios = require('axios');

const API_KEY = 'sk_live_6acv8fHhGrQTPQfVePKOYn_Yy1Wx8I3pRE75a4AwhF0';
const client = axios.create({
  baseURL: 'http://localhost:8085/api/v1',
  headers: { 'Authorization': `Bearer ${API_KEY}` }
});

async function runImport() {
  // Step 1: Start Session
  const { data: s } = await client.post('/session/start');
  const sessionId = s.session_id;

  // Step 2: Authenticate
  await client.post('/session/login', {
    session_id: sessionId,
    username: 'P25814',
    password: 'YourPassword',
    captcha: '9egB77'
  });

  // Step 3: Fetch Students
  const { data: records } = await client.post('/session/fetch-students', {
    session_id: sessionId,
    school_code: 'P25814',
    session_val: '2024-2025',
    class_val: '1'
  });

  console.log(`Extracted ${records.count} students:`, records.rows);

  // Step 4: Close Session
  await client.post('/session/close', { session_id: sessionId });
}

runImport().catch(console.error);

3. Headless REST API Endpoints Specification

Method Endpoint Description
GET /api/v1/account/status Check current plan, monthly request quota, and remaining credits.
POST /api/v1/session/start Create an isolated browser context and receive a base64 live CAPTCHA challenge.
POST /api/v1/session/refresh-captcha Refresh the live CAPTCHA image for an active session.
POST /api/v1/session/login Authenticate school credentials & CAPTCHA. Returns available Sessions and Classes.
POST /api/v1/session/fetch-students Extract student records for selected Session and Class. Deducts 1 quota credit.
POST /api/v1/automated/fetch-students Enterprise Tier: Single-call automated sync using local AI/OCR captcha solving.
POST /api/v1/session/close Explicitly release and destroy ephemeral browser context.

🛡️ Data Privacy & Zero-Retention Compliance

The RVS SAH platform is strictly a Pass-Through Data Pipeline. Student personal data (names, Aadhaar numbers, parents' details, mobile numbers) are never stored on SaaS servers. Extracted records are delivered straight to the client API response, fulfilling Indian Digital Personal Data Protection (DPDP) Act requirements.
